// Local-only packaging. Never builds source, uploads, contacts providers or changes the baseline.
import assert from 'node:assert/strict';import fs from 'node:fs/promises';import path from 'node:path';import {fileURLToPath} from 'node:url';import {hash,args,candidateGuard,canonicalPrivate,mapPayload,statusObject,parseConfig,committedConfig,loadStage,safeClass} from './status-ops-guards.mjs';
export async function packageVariants({baseline,candidate,sha,output}){
 const root=await candidateGuard(candidate,sha),base=await canonicalPrivate(baseline),destination=path.resolve(output);assert.ok(destination.startsWith('/private/tmp/flowboard-'));assert.equal(await fs.realpath(path.dirname(destination)),path.dirname(destination));assert.ok(destination!==base&&destination!==root&&!destination.startsWith(base+'/')&&!destination.startsWith(root+'/'));try{await fs.lstat(destination);throw new Error('OutputExists');}catch(error){if(error.code!=='ENOENT')throw error;}
 const original=await loadStage(base,sha);assert.equal(original.candidate,root);const before=await mapPayload(base);statusObject(await fs.readFile(base+'/out/release-status.json'),false);const configs={};for(const target of ['uat','production']){configs[target]=await committedConfig(root,sha,target);}assert.ok((await fs.readFile(base+'/wrangler.jsonc')).equals(configs.uat),'baseline must be exact UAT preparation');
 await fs.mkdir(destination,{mode:0o700});const variants={};
 for(const target of ['uat','production'])for(const active of [false,true]){const label=target+'-'+(active?'active':'inactive'),stage=destination+'/'+label;await fs.mkdir(stage,{mode:0o700});for(const name of ['out','functions'])await fs.cp(base+'/'+name,stage+'/'+name,{recursive:true,errorOnExist:true,force:false});await fs.writeFile(stage+'/wrangler.jsonc',configs[target],{flag:'wx'});if(active)await fs.writeFile(stage+'/out/release-status.json','{"active":true}\n');const pages=await mapPayload(stage);const artifact={...original,status:'packaged-not-deployed',sourceCommit:sha,candidate:root,pagesStage:stage,target,releaseStatus:{active},pages,pagesDigest:hash(JSON.stringify(pages)),baseline:{pagesStage:base,pagesDigest:original.pagesDigest,manifestSha256:hash(await fs.readFile(base+'/artifact-manifest.json'))},configurationOverlay:{status:{path:'out/release-status.json',active,baselineSha256:before['out/release-status.json'].sha256,sha256:pages['out/release-status.json'].sha256},wrangler:{path:'wrangler.jsonc',sourceCommittedPath:'apps/web/wrangler.'+(target==='uat'?'uat':'prod')+'.jsonc',sha256:pages['wrangler.jsonc'].sha256}},manifestMetadataIsNotPublishedPayload:true,createdAt:new Date().toISOString()};await fs.writeFile(stage+'/artifact-manifest.json',JSON.stringify(artifact,null,2)+'\n',{flag:'wx'});variants[label]={stage,pages,pagesDigest:artifact.pagesDigest,manifestSha256:hash(await fs.readFile(stage+'/artifact-manifest.json'))};}
 const differences={};for(const target of ['uat','production']){const off=variants[target+'-inactive'].pages,on=variants[target+'-active'].pages;assert.deepEqual(Object.keys(off),Object.keys(on));differences[target]=Object.keys(off).filter(k=>off[k].sha256!==on[k].sha256);assert.deepEqual(differences[target],['out/release-status.json']);}for(const active of ['active','inactive']){const u=variants['uat-'+active].pages,p=variants['production-'+active].pages;assert.deepEqual(Object.keys(u),Object.keys(p));assert.deepEqual(Object.keys(u).filter(k=>u[k].sha256!==p[k].sha256),['wrangler.jsonc']);}
 assert.deepEqual(await mapPayload(base),before,'baseline changed during package');await candidateGuard(root,sha);const receipt={status:'packaged-not-deployed',sourceCommit:sha,baseline:base,output:destination,variants,onlyChangedPublishedPathPerEnvironment:'out/release-status.json',sameEnvironmentStagedPayloadDifferences:differences,metadataExcludedFromPayloadComparison:['artifact-manifest.json'],workerManifestPreserved:true,applicationSourceMutated:false,cloudCalls:0};await fs.writeFile(destination+'/variant-manifest.json',JSON.stringify(receipt,null,2)+'\n',{flag:'wx'});return receipt;
}
export async function main(input=process.argv.slice(2)){try{const a=args(['execute','baseline','candidate','sha','output'],input);assert.equal(a.execute,'reviewed-local-packaging');const result=await packageVariants(a);console.log(JSON.stringify({status:result.status,sourceCommit:result.sourceCommit,output:result.output,variants:Object.keys(result.variants),cloudCalls:0}));return 0;}catch(error){console.log(JSON.stringify({status:'fail',errorClass:safeClass(error),cloudCalls:0,instruction:'STOP. Keep partial outputs for inspection; choose a new output path only after diagnosis.'}));return 1;}}
if(process.argv[1]&&path.resolve(process.argv[1])===fileURLToPath(import.meta.url))process.exitCode=await main();
