import test from 'node:test';import assert from 'node:assert/strict';import fs from 'node:fs/promises';import path from 'node:path';import {execFileSync} from 'node:child_process';import * as guards from './status-ops-guards.mjs';import {packageVariants} from './package-status-variants.mjs';
import {fixture,git} from './status-ops-fixture.mjs';
test('packages4variants with one exact status-file difference, outside-root metadata, expected config overlay and unchanged baseline',async()=>{const f=await fixture();try{const before=await guards.mapPayload(f.baseline),result=await packageVariants(f);assert.equal(result.status,'packaged-not-deployed');for(const target of ['uat','production']){const off=f.output+'/'+target+'-inactive',on=f.output+'/'+target+'-active';const a=await guards.mapPayload(off),b=await guards.mapPayload(on);assert.deepEqual(Object.keys(a),Object.keys(b));assert.deepEqual(Object.keys(a).filter(k=>a[k].sha256!==b[k].sha256),['out/release-status.json']);const manifest=await guards.loadStage(on,f.sha,target,true);assert.equal(manifest.releaseStatus.active,true);assert.equal(manifest.configurationOverlay.wrangler.sourceCommittedPath,'apps/web/wrangler.'+(target==='uat'?'uat':'prod')+'.jsonc');assert.equal(await fs.readFile(on+'/out/release-status.json','utf8'),'{"active":true}\n');}assert.deepEqual(await guards.mapPayload(f.baseline),before);await assert.rejects(packageVariants(f));}finally{await f.close();}});
test('wrongSHA, dirty or attached candidate, wrong base manifest/hash and symlinks fail before output creation',async()=>{for(const variant of ['sha','dirty','attached','hash','source','symlink']){const f=await fixture();try{if(variant==='sha')f.sha='a'.repeat(40);if(variant==='dirty')await fs.writeFile(f.candidate+'/untracked.txt','synthetic');if(variant==='attached')git(f.candidate,['checkout','-b','synthetic']);if(variant==='hash')await fs.writeFile(f.baseline+'/out/connect.html','tampered');if(variant==='source'){f.artifact.sourceCommit='b'.repeat(40);await fs.writeFile(f.baseline+'/artifact-manifest.json',JSON.stringify(f.artifact));}if(variant==='symlink')await fs.symlink('/nonexistent',f.baseline+'/out/unsafe');await assert.rejects(packageVariants(f));await assert.rejects(fs.lstat(f.output),{code:'ENOENT'});}finally{await f.close();}}});
test('invalid status schemas and wrong committed service configuration fail closed',async()=>{for(const bytes of ['{"active":"false"}','{"active":false,"message":"private"}','[]','{"active":true}']){const f=await fixture();try{await fs.writeFile(f.baseline+'/out/release-status.json',bytes);const pages=await guards.mapPayload(f.baseline);f.artifact.pages=pages;f.artifact.pagesDigest=guards.hash(JSON.stringify(pages));await fs.writeFile(f.baseline+'/artifact-manifest.json',JSON.stringify(f.artifact));await assert.rejects(packageVariants(f));}finally{await f.close();}}assert.throws(()=>guards.parseConfig('{"name":"flowboard-app","pages_build_output_dir":"./out","services":[{"binding":"API","service":"flowboard-api-uat"}]}','production'));});
test('request guard refuses foreign base, scheme-relative URLs, credential headers and path escapes',()=>{for(const args of [['//attacker.invalid',{},guards.ORIGINS.uat],['/x',{headers:{authorization:'synthetic'}},guards.ORIGINS.uat],['/x',{headers:{cookie:'synthetic'}},guards.ORIGINS.uat],['/x',{},'https://attacker.invalid']])assert.throws(()=>guards.requestOptions(...args,'uat'));const allowed=guards.requestOptions('/release-status.json',{},guards.ORIGINS.uat,'uat');assert.equal(allowed.url.origin,guards.ORIGINS.uat);assert.equal(allowed.init.credentials,'omit');assert.equal(allowed.init.redirect,'manual');});
test('report path must be new and canonically inside release folder',async()=>{await assert.rejects(guards.newReport('/private/tmp/outside-status.json'));await assert.rejects(guards.newReport(path.resolve('docs/reviews/PROD-MCP-20261106/PLAN.md')));});
test('config parser preserves quoted URLs while rejecting wrong bindings and stage metadata cannot excuse uncommitted config bytes',async()=>{assert.doesNotThrow(()=>guards.parseConfig('// comment\n{"name":"flowboard-app","pages_build_output_dir":"./out","services":[{"binding":"API","service":"flowboard-api"}],"extra":"https://example.invalid/*quoted*/"}/*end*/','production'));const f=await fixture();try{await packageVariants(f);const stage=f.output+'/production-inactive',manifest=JSON.parse(await fs.readFile(stage+'/artifact-manifest.json','utf8'));await fs.appendFile(stage+'/wrangler.jsonc','\n// unreviewed configuration byte drift\n');manifest.pages=await guards.mapPayload(stage);manifest.pagesDigest=guards.hash(JSON.stringify(manifest.pages));await fs.writeFile(stage+'/artifact-manifest.json',JSON.stringify(manifest));await assert.rejects(guards.loadStage(stage,f.sha,'production',false),/committed environment/);}finally{await f.close();}});
