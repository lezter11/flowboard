// Prepared bounded anonymous probe. No import-time I/O/network; root executes only after verified deployment.
import assert from 'node:assert/strict';import fs from 'node:fs/promises';import path from 'node:path';import {fileURLToPath} from 'node:url';
import {effectiveReferrerPolicy} from '../../HOSTED-MCP-20261104/release-dce1a67/referrer-policy.mjs';
import {ORIGINS,WORKERS,hash,args as parseArgs,canonicalPrivate,loadStage,newReport,requestOptions,statusObject,safeClass} from './status-ops-guards.mjs';
const MAX_REQUESTS=25,MAX_BYTES=4*1024*1024;
export async function runSmoke(args,{fetchImpl=globalThis.fetch,writeReport}={}){
 assert.equal(args.execute,'reviewed-deployment');assert.ok(Object.hasOwn(ORIGINS,args.target));assert.match(args.sha??'',/^[a-f0-9]{40}$/);assert.ok(args['status-active']==='true'||args['status-active']==='false');
 for(const key of ['worker-version','pages-deployment'])assert.match(args[key]??'',/^[a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12}$/);
 const COMMIT=args.sha,STAGE=await canonicalPrivate(args.stage),EXPECTED_ACTIVE=args['status-active']==='true',reportFile=await newReport(args.report);
 const artifact=await loadStage(STAGE,COMMIT,args.target,EXPECTED_ACTIVE),ORIGIN=ORIGINS[args.target],WORKER=WORKERS[args.target],METADATA=ORIGIN+'/.well-known/oauth-protected-resource/api/v1/mcp';
 const localAssets=Object.keys(artifact.pages);assert.ok(localAssets.some(x=>x.includes('/app/settings/')&&x.endsWith('.js'))&&localAssets.some(x=>x.startsWith('out/_next/static/')&&x.endsWith('.css'))&&artifact.pages['out/connect.html']&&artifact.pages['out/release-status.json']);
const report={kind:'bounded anonymous '+args.target+' hosted connector smoke',sourceCommit:COMMIT,origin:ORIGIN,workerVersion:args['worker-version']??null,pagesDeployment:args['pages-deployment']??null,startedAt:new Date().toISOString(),maxRequests:MAX_REQUESTS,requestCount:0,results:[],notCovered:[{id:'signed-in-google-claude',reason:'Anonymous smoke cannot prove provider login, consent, token exchange, authenticated tools or refresh/disconnect.'}],credentialsUsed:false,applicationMutations:false};
const noStore=r=>/\bno-store\b/i.test(r.headers.get('cache-control')??'');
const noCredentials=r=>!r.headers.has('access-control-allow-credentials');
async function request(urlPath,init={},base=ORIGIN){
  const verified=requestOptions(urlPath,init,base,args.target),url=verified.url;assert.ok(report.requestCount<MAX_REQUESTS,'hard request cap reached');
  report.requestCount++;const r=await fetchImpl(url,verified.init);
  const declared=r.headers.get('content-length');assert.ok(declared===null||Number(declared)<=MAX_BYTES,'response cap exceeded');let bytes=0;const parts=[];
  if(r.body){const reader=r.body.getReader();try{while(true){const {done,value}=await reader.read();if(done)break;bytes+=value.byteLength;if(bytes>MAX_BYTES){void reader.cancel().catch(()=>{});throw new Error('ResponseCap');}parts.push(value);}}finally{reader.releaseLock();}}
  const body=Buffer.concat(parts);return {r,body,json:()=>{try{return JSON.parse(body.toString('utf8'));}catch{return null;}}};
}
async function check(id,fn){try{const result=await fn();report.results.push({id,...result,pass:Object.values(result.checks).every(v=>v===true)});}catch(e){report.results.push({id,pass:false,errorClass:safeClass(e)});}}
for(const [id,p] of [['resource-specific','/.well-known/oauth-protected-resource/api/v1/mcp'],['resource-root','/.well-known/oauth-protected-resource']])await check(id,async()=>{const {r,json}=await request(p);const j=json();return {status:r.status,checks:{status200:r.status===200,json:/application\/json/i.test(r.headers.get('content-type')??''),noStore:noStore(r),notRedirect:!r.headers.has('location'),exactResource:j?.resource===ORIGIN+'/api/v1/mcp',exactIssuer:Array.isArray(j?.authorization_servers)&&j.authorization_servers.length===1&&j.authorization_servers[0]===ORIGIN,headerBearer:j?.bearer_methods_supported?.length===1&&j.bearer_methods_supported[0]==='header'}};});
await check('authorization-server',async()=>{const {r,json}=await request('/.well-known/oauth-authorization-server');const j=json();return {status:r.status,checks:{status200:r.status===200,json:/application\/json/i.test(r.headers.get('content-type')??''),noStore:noStore(r),notRedirect:!r.headers.has('location'),exactIssuer:j?.issuer===ORIGIN,exactAuthorize:j?.authorization_endpoint===ORIGIN+'/api/v1/mcp/oauth/authorize',exactToken:j?.token_endpoint===ORIGIN+'/api/v1/mcp/oauth/token',exactRegister:j?.registration_endpoint===ORIGIN+'/api/v1/mcp/oauth/register',exactRevoke:j?.revocation_endpoint===ORIGIN+'/api/v1/mcp/oauth/revoke',pkce:j?.code_challenge_methods_supported?.includes('S256')===true,cimd:j?.client_id_metadata_document_supported===true}};});
for(const method of ['POST','GET','DELETE'])await check('anonymous-mcp-'+method.toLowerCase(),async()=>{const {r,json}=await request('/api/v1/mcp',{method,headers:{accept:'application/json, text/event-stream',...(method==='POST'?{'content-type':'application/json'}:{})},...(method==='POST'?{body:JSON.stringify({jsonrpc:'2.0',id:1,method:'ping'})}:{})});const j=json();return {status:r.status,checks:{status401:r.status===401,noStore:noStore(r),exactChallenge:r.headers.get('www-authenticate')===`Bearer resource_metadata="${METADATA}"`,noCredentials:noCredentials(r),noCookie:!r.headers.has('set-cookie'),errorOnly:j?.jsonrpc==='2.0'&&j?.error!==undefined&&j?.result===undefined&&j?.data===undefined}};});
await check('claude-preflight',async()=>{const {r,body}=await request('/api/v1/mcp',{method:'OPTIONS',headers:{origin:'https://claude.ai','access-control-request-method':'POST','access-control-request-headers':'authorization,content-type,accept,mcp-protocol-version'}});return {status:r.status,checks:{status204:r.status===204,noStore:noStore(r),exactOrigin:r.headers.get('access-control-allow-origin')==='https://claude.ai',noCredentials:noCredentials(r),noCookie:!r.headers.has('set-cookie'),empty:body.length===0,postAllowed:(r.headers.get('access-control-allow-methods')??'').split(',').map(x=>x.trim()).includes('POST')}};});
await check('attacker-origin',async()=>{const {r,json}=await request('/api/v1/mcp',{method:'OPTIONS',headers:{origin:'https://attacker.invalid','access-control-request-method':'POST'}});const j=json();return {status:r.status,checks:{status403:r.status===403,noStore:noStore(r),noAllowOrigin:!r.headers.has('access-control-allow-origin'),noCredentials:noCredentials(r),noCookie:!r.headers.has('set-cookie'),noData:j?.result===undefined&&j?.data===undefined}};});
for(const [id,p] of [['own-connections','/api/v1/mcp/connections'],['session','/api/v1/session'],['tokens','/api/v1/tokens']])await check(id,async()=>{const {r,json}=await request(p,{headers:{accept:'application/json'}});const j=json();return {status:r.status,checks:{status401:r.status===401,noStore:noStore(r),noData:j!==null&&j?.data===undefined&&j?.error!==undefined,notRedirect:!r.headers.has('location')}};});
await check('connect-invalid-handle-static',async()=>{const {r,body}=await request('/connect?request=invalid');const local=await fs.readFile(STAGE+'/out/connect.html');assert.equal(hash(local),artifact.pages['out/connect.html'].sha256);return {status:r.status,bytes:body.length,sha256:hash(body),checks:{status200:r.status===200,html:/text\/html/i.test(r.headers.get('content-type')??''),noStore:noStore(r),noReferrer:effectiveReferrerPolicy(r.headers.get('referrer-policy'))==='no-referrer',exactPreparedHtml:body.equals(local),notRedirect:!r.headers.has('location')}};});
await check('connections-safe-signin',async()=>{const {r}=await request('/settings?tab=connections');let correct=false;try{const to=new URL(r.headers.get('location')??'',ORIGIN);correct=to.origin===ORIGIN&&to.pathname==='/signin'&&to.searchParams.get('next')==='/settings?tab=connections'&&[...to.searchParams.keys()].every(x=>x==='next');}catch{}return {status:r.status,checks:{status302:r.status===302,noStore:noStore(r),safeNext:correct}};});
const assetNames=Object.keys(artifact.pages).filter(x=>x.startsWith('out/_next/static/')&&(x.endsWith('.js')||x.endsWith('.css'))&&artifact.pages[x].bytes<1024*1024).sort();
const js=assetNames.find(x=>x.includes('/app/settings/')&&x.endsWith('.js'));const css=assetNames.find(x=>x.endsWith('.css'));assert.ok(js&&css);
for(const [id,name] of [['static-settings-js',js],['static-css',css]])await check(id,async()=>{const local=await fs.readFile(STAGE+'/'+name);assert.equal(hash(local),artifact.pages[name].sha256);const {r,body}=await request('/'+name.slice(4));return {status:r.status,bytes:body.length,sha256:hash(body),checks:{status200:r.status===200,exactPreparedBytes:body.equals(local),samplePrompt:id!=='static-settings-js'||body.includes(Buffer.from('Copy sample prompt')),immutable:/immutable/i.test(r.headers.get('cache-control')??''),notRedirect:!r.headers.has('location')}};});
for(const [name,p,expected] of [['live','/health/live','ok'],['ready','/health/ready','ready']])await check('health-'+name,async()=>{const {r,json}=await request(p,{headers:{accept:'application/json'}},WORKER);const j=json();return {status:r.status,checks:{status200:r.status===200,exactStatus:j?.data?.status===expected,json:/application\/json/i.test(r.headers.get('content-type')??''),notRedirect:!r.headers.has('location')}};});
await check('static-release-status',async()=>{const local=await fs.readFile(STAGE+'/out/release-status.json');assert.equal(hash(local),artifact.pages['out/release-status.json'].sha256);statusObject(local,EXPECTED_ACTIVE);const {r,body,json}=await request('/release-status.json');const value=json();return {status:r.status,bytes:body.length,sha256:hash(body),checks:{status200:r.status===200,json:/application\/json/i.test(r.headers.get('content-type')??''),noStore:noStore(r),exactPreparedBytes:body.equals(local),exactActiveObject:value!==null&&typeof value==='object'&&!Array.isArray(value)&&Object.keys(value).length===1&&Object.keys(value)[0]==='active'&&typeof value.active==='boolean'&&value.active===EXPECTED_ACTIVE,noCookie:!r.headers.has('set-cookie'),noCredentials:noCredentials(r),notRedirect:!r.headers.has('location')}};});
report.completedAt=new Date().toISOString();report.passed=report.results.filter(r=>r.pass).length;report.failed=report.results.length-report.passed;report.exitCode=report.failed?1:0;report.status=report.failed?'fail':'anonymous-checks-pass-hosted-acceptance-pending';
if(writeReport)await writeReport(reportFile,report);else await fs.writeFile(reportFile,JSON.stringify(report,null,2)+'\n',{flag:'wx',mode:0o600});
return report;

}
export async function main(input=process.argv.slice(2)){try{const a=parseArgs(['execute','report','worker-version','pages-deployment','target','sha','stage','status-active'],input);const r=await runSmoke(a);console.log(JSON.stringify({status:r.status,sourceCommit:r.sourceCommit,requestCount:r.requestCount,passed:r.passed,failed:r.failed,exitCode:r.exitCode}));return r.exitCode;}catch(error){console.log(JSON.stringify({status:'fail',errorClass:safeClass(error),instruction:'STOP. Missing or incomplete receipt is not a pass.'}));return 1;}}
if(process.argv[1]&&path.resolve(process.argv[1])===fileURLToPath(import.meta.url))process.exitCode=await main();
