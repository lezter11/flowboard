import test from 'node:test';import assert from 'node:assert/strict';import fs from 'node:fs/promises';import path from 'node:path';import {randomUUID} from 'node:crypto';import {fixture} from './status-ops-fixture.mjs';import {packageVariants} from './package-status-variants.mjs';import {runSmoke} from './anonymous-release-smoke.mjs';import {ORIGINS,WORKERS} from './status-ops-guards.mjs';
function parameters(f,target,active){return {execute:'reviewed-deployment',target,sha:f.sha,stage:f.output+'/'+target+'-'+(active?'active':'inactive'),'status-active':String(active),report:path.resolve('docs/reviews/PROD-MCP-20261106/ops-preflight/unused-synthetic-'+randomUUID()+'.json'),'worker-version':'12345678-1234-1234-1234-123456789abc','pages-deployment':'abcdef01-2345-6789-abcd-ef0123456789'};}
function mockFetch(stage,target,overrides={}){const calls=[],origin=ORIGINS[target];const fetchImpl=async(url,init)=>{assert.ok(url.origin===origin||url.origin===WORKERS[target]);assert.equal(init.redirect,'manual');assert.equal(init.credentials,'omit');assert.equal(init.headers.has('authorization'),false);assert.equal(init.headers.has('cookie'),false);calls.push({origin:url.origin,path:url.pathname,method:init.method??'GET'});if(overrides[url.pathname])return overrides[url.pathname](url,init);const headers={'cache-control':'no-store','content-type':'application/json'};const json=(body,status=200,extra={})=>new Response(JSON.stringify(body),{status,headers:{...headers,...extra}});if(url.pathname.startsWith('/.well-known/oauth-protected-resource'))return json({resource:origin+'/api/v1/mcp',authorization_servers:[origin],bearer_methods_supported:['header']});if(url.pathname==='/.well-known/oauth-authorization-server')return json({issuer:origin,authorization_endpoint:origin+'/api/v1/mcp/oauth/authorize',token_endpoint:origin+'/api/v1/mcp/oauth/token',registration_endpoint:origin+'/api/v1/mcp/oauth/register',revocation_endpoint:origin+'/api/v1/mcp/oauth/revoke',code_challenge_methods_supported:['S256'],client_id_metadata_document_supported:true});if(url.pathname==='/api/v1/mcp'){if(init.method==='OPTIONS')return init.headers.get('origin')==='https://claude.ai'?new Response(null,{status:204,headers:{...headers,'access-control-allow-origin':'https://claude.ai','access-control-allow-methods':'POST'}}):json({error:'forbidden'},403);if(init.method==='POST')assert.equal(JSON.parse(init.body).method,'ping');return json({jsonrpc:'2.0',error:{code:-32000}},401,{'www-authenticate':`Bearer resource_metadata="${origin}/.well-known/oauth-protected-resource/api/v1/mcp"`});}if(['/api/v1/mcp/connections','/api/v1/session','/api/v1/tokens'].includes(url.pathname))return json({error:{code:'UNAUTHENTICATED'}},401);if(url.pathname==='/settings')return new Response(null,{status:302,headers:{...headers,location:origin+'/signin?next=%2Fsettings%3Ftab%3Dconnections'}});if(url.pathname==='/connect')return new Response(await fs.readFile(stage+'/out/connect.html'),{headers:{...headers,'content-type':'text/html','referrer-policy':'strict-origin-when-cross-origin, no-referrer'}});if(url.pathname.startsWith('/_next/static/'))return new Response(await fs.readFile(stage+'/out'+url.pathname),{headers:{'cache-control':'public, max-age=31536000, immutable'}});if(url.pathname==='/health/live'||url.pathname==='/health/ready'){assert.equal(url.origin,WORKERS[target]);return json({data:{status:url.pathname.endsWith('ready')?'ready':'ok'}});}if(url.pathname==='/release-status.json')return new Response(await fs.readFile(stage+'/out/release-status.json'),{headers});throw new Error('Unexpected synthetic request');};return {calls,fetchImpl};}
test('all4variants run exact18-request synthetic smoke with all prior assertions and statusbytes/no-store',async()=>{const f=await fixture();try{await packageVariants(f);for(const target of ['uat','production'])for(const active of [false,true]){const a=parameters(f,target,active),mock=mockFetch(a.stage,target);let saved;const r=await runSmoke(a,{fetchImpl:mock.fetchImpl,writeReport:async(_file,value)=>{saved=value;}});assert.equal(r.requestCount,18);assert.equal(mock.calls.length,18);assert.equal(r.passed,18);assert.equal(r.failed,0);assert.equal(r.exitCode,0);assert.equal(saved,r);assert.equal(r.results.find(x=>x.id==='static-release-status').pass,true);assert.equal(r.results.find(x=>x.id==='connect-invalid-handle-static').checks.noReferrer,true);assert.deepEqual(r.results.slice(0,17).map(x=>x.id),['resource-specific','resource-root','authorization-server','anonymous-mcp-post','anonymous-mcp-get','anonymous-mcp-delete','claude-preflight','attacker-origin','own-connections','session','tokens','connect-invalid-handle-static','connections-safe-signin','static-settings-js','static-css','health-live','health-ready']);await assert.rejects(fs.lstat(a.report),{code:'ENOENT'});}}finally{await f.close();}});
