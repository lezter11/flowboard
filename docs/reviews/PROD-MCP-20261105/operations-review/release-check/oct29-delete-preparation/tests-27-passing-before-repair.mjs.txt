import test from 'node:test';
import assert from 'node:assert/strict';
import { TARGET, PRESERVED, CANDIDATE, DELETE_COMMAND, assertGates, validateBefore, verifyAfter, runApprovedDeletion } from './delete-oct29-uat-backup.mjs';
const approval='delete-only-br-damp-poetry-a1gfbcvn-and-ep-bold-hill-a1mlmaly';
function fixture(){const branches=[...PRESERVED.map(x=>({...x,current_state:'ready'})),{...TARGET,current_state:'ready',primary:false,default:false,protected:false},{id:'br-synthetic-other',name:'other-backup',parent_id:TARGET.parent_id}];const endpoints=[{id:TARGET.endpointId,branch_id:TARGET.id,current_state:'idle',type:'read_write'},{id:'ep-patient-frost-a1kvcqki',branch_id:PRESERVED[0].id,type:'read_write',current_state:'active'},{id:'ep-shy-fire-a1we7145',branch_id:PRESERVED[1].id,type:'read_write',current_state:'active'}];return {branches,endpoints};}
function harness(change=()=>{}){const before=fixture();change(before);let deleted=false,deleteCalls=0,readCalls=0;const deps={async listBranches(){readCalls++;return deleted?before.branches.filter(b=>b.id!==TARGET.id):structuredClone(before.branches);},async listEndpoints(){return deleted?before.endpoints.filter(e=>e.id!==TARGET.endpointId):structuredClone(before.endpoints);},async deleteExact(command){assert.deepEqual(command,DELETE_COMMAND);deleteCalls++;deleted=true;},async pause(){}};return {before,deps,get deleteCalls(){return deleteCalls;},get readCalls(){return readCalls;}};}
const gates={execute:'owner-approved-single-delete',ownerApproval:approval,uatGate:'accepted-'+CANDIDATE};
test('entry requires exact execute, separate owner approval and exact-source UAT gate before any provider read',async()=>{for(const bad of [{...gates,execute:''},{...gates,ownerApproval:'old-oct28-approval'},{...gates,uatGate:'accepted-other'}]){const h=harness();const r=await runApprovedDeletion(bad,h.deps);assert.equal(r.status,'fail');assert.equal(h.readCalls,0);assert.equal(h.deleteCalls,0);}assert.doesNotThrow(()=>assertGates(gates));});
test('command is fixed to exactly one branch and project; no target argument',()=>{assert.deepEqual(DELETE_COMMAND,['branches','delete','br-damp-poetry-a1gfbcvn','--project-id','misty-unit-76664549']);assert.equal(Object.isFrozen(DELETE_COMMAND),true);});
for(const [name,change]of [['wrong target ID',f=>f.branches.find(b=>b.id===TARGET.id).id='br-other-target'],['wrong name',f=>f.branches.find(b=>b.id===TARGET.id).name='wrong'],['wrong parent',f=>f.branches.find(b=>b.id===TARGET.id).parent_id='br-other-parent'],['has child',f=>f.branches.push({id:'br-synthetic-child',parent_id:TARGET.id})],['primary',f=>f.branches.find(b=>b.id===TARGET.id).primary=true],['default',f=>f.branches.find(b=>b.id===TARGET.id).default=true],['protected',f=>f.branches.find(b=>b.id===TARGET.id).protected=true],['missing safety flag',f=>delete f.branches.find(b=>b.id===TARGET.id).protected],['not ready',f=>f.branches.find(b=>b.id===TARGET.id).current_state='init'],['preserved branch missing',f=>f.branches.splice(0,1)],['preserved backup renamed',f=>f.branches.find(b=>b.id===PRESERVED[2].id).name='wrong'],['wrong endpoint',f=>f.endpoints[0].id='ep-other'],['endpoint wrong parent',f=>f.endpoints[0].branch_id='br-other'],['compute not idle',f=>f.endpoints[0].current_state='active'],['wrong compute type',f=>f.endpoints[0].type='read_only'],['compute missing',f=>f.endpoints.shift()],['target marked deleted',f=>f.branches.find(b=>b.id===TARGET.id).deleted_at='synthetic'],['extra compute',f=>f.endpoints.push({id:'ep-extra',branch_id:TARGET.id,current_state:'idle',type:'read_write'})],['duplicate branch IDs',f=>f.branches.push({...f.branches[0]})],['active endpoint missing',f=>f.endpoints.splice(1,1)]])test('refuses '+name+' without deletion',async()=>{const h=harness(change),r=await runApprovedDeletion(gates,h.deps);assert.equal(r.status,'fail');assert.equal(h.deleteCalls,0);});
test('each active branch and both production backups must independently remain present',async()=>{for(const preserved of PRESERVED){const h=harness(f=>{f.branches=f.branches.filter(b=>b.id!==preserved.id);});const r=await runApprovedDeletion(gates,h.deps);assert.equal(r.status,'fail');assert.equal(h.deleteCalls,0);}});
test('success preserves every other branch and endpoint, deletes once, records only safe IDs/counts',async()=>{const h=harness(),r=await runApprovedDeletion(gates,h.deps);assert.equal(r.status,'pass');assert.equal(r.targetAbsent,true);assert.equal(r.allOtherBranchesPreserved,true);assert.equal(r.allOtherEndpointsPreserved,true);assert.equal(h.deleteCalls,1);assert.equal(r.beforeBranchCount-r.afterBranchCount,1);assert.deepEqual(r.preservedBranchIds,h.before.branches.filter(b=>b.id!==TARGET.id).map(b=>b.id).sort());});
test('exact before/after sets reject any other removed or added branch or endpoint',()=>{const before=fixture(),state=validateBefore(before.branches,before.endpoints);const after=before.branches.filter(b=>b.id!==TARGET.id),endpoints=before.endpoints.filter(e=>e.id!==TARGET.endpointId);assert.doesNotThrow(()=>verifyAfter(after,endpoints,state));for(const a of [after.slice(1),[...after,{id:'br-unexpected'}]])assert.throws(()=>verifyAfter(a,endpoints,state));assert.throws(()=>verifyAfter(after,[],state));assert.throws(()=>verifyAfter(after,[...endpoints,{id:'ep-added',branch_id:PRESERVED[0].id}],state));assert.throws(()=>verifyAfter(after,endpoints.map(e=>({...e,branch_id:'br-swapped'})),state));});
test('unconfirmed/lost-response deletion failure stops without retry/fallback and never returns child diagnostics',async()=>{const h=harness();let calls=0;h.deps.deleteExact=async(command)=>{calls++;assert.deepEqual(command,DELETE_COMMAND);throw Object.assign(new Error('SYNTHETIC_PRIVATE postgres://secret'),{name:'SYNTHETIC_PRIVATE',stderr:'SYNTHETIC_PRIVATE',stdout:'SYNTHETIC_PRIVATE',code:'SYNTHETIC_PRIVATE',status:7});};const r=await runApprovedDeletion(gates,h.deps);assert.equal(r.status,'fail');assert.equal(r.mutationRequested,true);assert.equal(r.stage,'delete-approved-target');assert.equal(r.childExit,7);assert.equal(calls,1);assert.equal(JSON.stringify(r).includes('SYNTHETIC_PRIVATE'),false);assert.equal(JSON.stringify(r).includes('postgres://'),false);});
test('verification timeout never repeats delete or changes target',async()=>{const h=harness();let calls=0;h.deps.deleteExact=async(command)=>{assert.deepEqual(command,DELETE_COMMAND);calls++;};const r=await runApprovedDeletion(gates,h.deps);assert.equal(r.status,'fail');assert.equal(calls,1);assert.equal(r.stage,'verify-deleted');});
